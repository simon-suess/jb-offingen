# Rate limiting: 10 req/s per IP, burst 100, slow down excess
limit_req_zone $binary_remote_addr zone=jbo_zone:10m rate=10r/s;

server {
    server_name jb-offingen.de www.jb-offingen.de;

    location / {
        limit_req zone=jbo_zone burst=100 nodelay;
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }

    # Security headers
    add_header Strict-Transport-Security "max-age=63072000" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Permissions-Policy "geolocation=(), microphone=(), camera=()" always;

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/jb-offingen.de/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/jb-offingen.de/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}

server {
    if ($host = www.jb-offingen.de) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    if ($host = jb-offingen.de) {
        return 301 https://$host$request_uri;
    } # managed by Certbot

    listen 80;
    listen [::]:80;
    server_name jb-offingen.de www.jb-offingen.de;
    return 404; # managed by Certbot

}
